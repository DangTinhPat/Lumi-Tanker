cmake_minimum_required(VERSION 3.20)
project(tap_sim)
find_package(ament_cmake REQUIRED)
find_package(gz_ros2_control REQUIRED)
find_package(pluginlib REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(tap_contact_snapshot SHARED src/contact_snapshot.cpp)
target_compile_features(tap_contact_snapshot PUBLIC cxx_std_17)
target_link_libraries(tap_contact_snapshot gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::gz-transport13)
install(TARGETS tap_contact_snapshot LIBRARY DESTINATION lib)
add_library(tap_bounded_gazebo_system SHARED src/bounded_gazebo_system.cpp)
target_compile_features(tap_bounded_gazebo_system PUBLIC cxx_std_17)
target_include_directories(tap_bounded_gazebo_system PRIVATE include)
ament_target_dependencies(tap_bounded_gazebo_system gz_ros2_control pluginlib hardware_interface rclcpp std_msgs sensor_msgs)
target_link_libraries(tap_bounded_gazebo_system gz-sim8::gz-sim8)
pluginlib_export_plugin_description_file(gz_ros2_control bounded_hardware_plugins.xml)
install(TARGETS tap_bounded_gazebo_system LIBRARY DESTINATION lib)
install(DIRECTORY launch config worlds DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/activate_controllers DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_actuator_bounds tests/test_actuator_bounds.cpp)
  target_include_directories(test_actuator_bounds PRIVATE include)
  target_compile_features(test_actuator_bounds PRIVATE cxx_std_17)
  add_test(NAME actuator_bounds COMMAND test_actuator_bounds)
endif()
ament_package()
